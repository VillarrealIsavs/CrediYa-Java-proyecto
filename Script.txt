CREATE DATABASE crediya;
USE crediya;

SELECT id, estado, total
FROM prestamos
WHERE id = 1;

ALTER TABLE prestamos
ADD COLUMN fecha_vencimiento DATE;

UPDATE prestamos
SET fecha_vencimiento = DATE_ADD(fecha_inicio, INTERVAL cuotas MONTH)
WHERE id > 0;

CREATE TABLE empleados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50),
    documento VARCHAR(30),
    rol VARCHAR(30),
    correo VARCHAR(80),
    salario DOUBLE
);

CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50),
    documento VARCHAR(30),
    correo VARCHAR(80),
    telefono VARCHAR(20)
);

CREATE TABLE prestamos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id INT,
    empleado_id INT,
    monto DOUBLE,
    interes DOUBLE,
    cuotas INT,
    fecha_inicio DATE,
    estado VARCHAR(20),
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (empleado_id) REFERENCES empleados(id)
);

CREATE TABLE pagos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    prestamo_id INT,
    monto DOUBLE,
    fecha DATE,
    FOREIGN KEY (prestamo_id) REFERENCES prestamos(id)
);
